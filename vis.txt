  dlg.classList.toggle("show",show);
  if(show)setTimeout(()=>dlg.querySelector("[data-close-dlg]").focus(),80);
}
$$("[data-close-dlg]").forEach(b=>b.addEventListener("click",()=>toggleDlg(false)));

/* ════════════════ CONTROLS: seg / switch / slider ══════════ */
$$(".seg").forEach(g=>{
  const btns=[...g.querySelectorAll("button")];
  const select=b=>{btns.forEach(x=>x.setAttribute("aria-checked",x===b));
    if(g.id==="cModeSeg"){state.mode=b.dataset.val;renderHero();readout();}};
  btns.forEach(b=>b.addEventListener("click",()=>select(b)));
  g.addEventListener("keydown",e=>{
    if(!["ArrowLeft","ArrowRight","ArrowUp","ArrowDown"].includes(e.key))return;
    e.preventDefault();
    const i=btns.findIndex(b=>b.getAttribute("aria-checked")==="true");
    const n=(i+(e.key==="ArrowRight"||e.key==="ArrowDown"?1:-1)+btns.length)%btns.length;
    select(btns[n]);btns[n].focus();
  });
});
$$(".switch").forEach(s=>{
  const flip=()=>s.setAttribute("aria-checked",s.getAttribute("aria-checked")!=="true");

});
function syncSlider(input,chip,fmt){
  const min=+input.min,max=+input.max,v=+input.value,p=((v-min)/(max-min))*100;
  input.style.setProperty("--p",p+"%");
  chip.textContent=fmt?fmt(v):v;
  chip.style.left=`calc(${p}% + ${(50-p)*0.2}px)`;
  chip.classList.remove("pop");void chip.offsetWidth;chip.classList.add("pop");
}

/* ════════════════════════ KEYBOARD ═════════════════════════ */
addEventListener("keydown",e=>{
  if(e.target.matches("input,textarea"))return;
  const k=e.key.toLowerCase();
  const kb=$(`kbd[data-key="${k==="?"?"?":k}"]`);
  if(kb){kb.classList.add("lit");setTimeout(()=>kb.classList.remove("lit"),260);}
  if(k==="escape"){if(dlg.classList.contains("show"))toggleDlg(false);else if(sheetOpen)closeSheet();}
  else if(k==="s"&&!e.metaKey&&!e.ctrlKey){sheetOpen?closeSheet():openSheet();}
  else if(k==="r"&&!e.metaKey&&!e.ctrlKey){e.preventDefault();refresh();}
  else if(e.key==="?"){toggleDlg();}
});
